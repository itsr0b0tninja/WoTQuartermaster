def obj(value):
    return value if isinstance(value, dict) else {}


def atomic_text(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + '.tmp')
    tmp.write_text(value, encoding='utf-8')
    os.replace(tmp, path)


def absolute(base, value):
    p = Path(value).expanduser()
    return (p if p.is_absolute() else base / p).resolve()


def safe_child(root, relative):
    # Game manifests use Windows separators, even when inspected on Linux.
    if not isinstance(relative, str) or not relative:
        raise ValueError('Missing relative path')
    relative = relative.replace('\\', '/')
    p = Path(relative)
    if p.is_absolute() or ':' in relative or '..' in p.parts:
        raise ValueError('Unsafe catalogue relative path')
    result = (root / p).resolve()
    if not result.is_relative_to(root.resolve()):
        raise ValueError('Path escapes catalogue')
    return result


def read_json(path):
    with path.open('r', encoding='utf-8-sig') as f:
        data = json.load(f)
    if not isinstance(data, dict):
        raise ValueError('Expected JSON object: ' + path.name)
    return data


def verified_bytes(path, expected):
    raw = path.read_bytes()
    if not isinstance(expected, str) or hashlib.sha256(raw).hexdigest() != expected:
        raise ValueError('Checksum mismatch: ' + path.name)
    return raw


def number(value):
    return f'{value:,}' if isinstance(value, int) and not isinstance(value, bool) else ('—' if value is None else str(value))


def age_of(payload, now):
    try:
        value = now - float(payload['heartbeat_at'])
        return max(0., value) if math.isfinite(value) and value >= -5 else float('inf')
    except (KeyError, ValueError, TypeError, OverflowError):
        return float('inf')


def valid_context(c):
    return (c.get('schema') == 1 and isinstance(c.get('session'), str) and
            c.get('connector_version') == CONNECTOR_VERSION and c.get('state') in ACTIVE_BATTLE and
            c.get('map_ready') is True and c.get('battle_id') is not None and
            c.get('map_id') is not None)


def connection(conn, ctx, conn_disk, ctx_disk, now=None):
    now = time.time() if now is None else now
    if not conn:
        return 'missing', 'Connector missing', None
    if conn.get('schema') != 1 or conn.get('connector_version') != CONNECTOR_VERSION:
        return 'version', 'Connector version mismatch', None
    if conn.get('game_state') == 'stopped' or conn.get('status') == 'stopped':
        return 'closed', 'Game closed · saved data', None
    # Context is faster and authoritative for the second-monitor state. Never use
    # cached context as live, and never mix sessions or game versions.
    same = (ctx.get('session') == conn.get('session') and ctx.get('game_version') == conn.get('game_version'))
    recent = ctx_disk and same and ctx.get('schema') == 1 and ctx.get('connector_version') == CONNECTOR_VERSION and age_of(ctx, now) <= 10
    if not conn_disk or age_of(conn, now) > 30:
        return 'stale', 'Stale connection · saved data', None
    if recent:
        state = ctx.get('state')
        if state == 'stopped':
            return 'closed', 'Game closed · saved data', None
        if state in ACTIVE_BATTLE:
            if valid_context(ctx):
                return state, 'Connected · ' + ('Loading battle' if state == 'battle_loading' else 'In battle'), ctx
            return state, 'Connected · waiting for map', None
        if state in ('garage', 'matchmaking'):
            return state, 'Connected · ' + ('Garage' if state == 'garage' else 'Matchmaking'), None
        return 'waiting', 'Connected · waiting for garage', None
    # Do not fall back to the last observed map or stale context.
    return 'stale', 'Live battle context unavailable · saved data', None


class Settings:
    defaults = {'game_path': r'C:\Games\World_of_Tanks_EU', 'data_path': 'data', 'playstyle': 'Hybrid scout',
                'strategy': 'Free-to-play', 'auto_switch': 'yes', 'remember_window': 'yes', 'always_on_top': 'no',

        self.base = base
        self.path = base / 'settings.ini'
        self.values = self.defaults.copy()
        self.warning = ''
        if self.path.exists():

                p = configparser.ConfigParser(interpolation=None)

class Store:
    """A disposable derived local cache. Source exports are never modified."""
    def __init__(self, root):
        self.root = root
        self.folder = root / 'database'
        self.folder.mkdir(parents=True, exist_ok=True)
        self.path = self.folder / 'quartermaster.sqlite3'
        self.db = sqlite3.connect(self.path, timeout=2)
        try:
            version = self.db.execute('PRAGMA user_version').fetchone()[0]
            if version not in (0, DB_SCHEMA):
                raise ValueError('Unsupported desktop database schema: ' + str(version))
            with self.db:
                self.db.execute('CREATE TABLE IF NOT EXISTS snapshots (source TEXT PRIMARY KEY, payload TEXT NOT NULL, observed REAL NOT NULL)')
                self.db.execute('CREATE TABLE IF NOT EXISTS map_records (arena_id TEXT PRIMARY KEY, payload TEXT NOT NULL)')
                self.db.execute('CREATE TABLE IF NOT EXISTS metadata (key TEXT PRIMARY KEY, value TEXT NOT NULL)')
                self.db.execute('CREATE TABLE IF NOT EXISTS garage_inventory (inventory_id TEXT PRIMARY KEY, payload TEXT NOT NULL)')
                self.db.execute('CREATE TABLE IF NOT EXISTS garage_sections (section TEXT PRIMARY KEY, payload TEXT NOT NULL)')
                self.db.execute(f'PRAGMA user_version={DB_SCHEMA}')
        except Exception:
            self.db.close(); raise
    def get(self, key):
        row = self.db.execute('SELECT payload FROM snapshots WHERE source=?', (key,)).fetchone()
        return json.loads(row[0]) if row else {}
    def put(self, key, value):
        with self.db:
            self.db.execute('INSERT OR REPLACE INTO snapshots VALUES (?,?,?)', (key, json.dumps(value, ensure_ascii=True), time.time()))
    def maps(self, rows, identity):
        encoded = json.dumps(identity, sort_keys=True)
        prior = self.db.execute("SELECT value FROM metadata WHERE key='map_identity'").fetchone()
        if prior and prior[0] == encoded:
            return
        with self.db:
            self.db.execute('DELETE FROM map_records')
            self.db.executemany('INSERT INTO map_records VALUES (?,?)', ((str(k), json.dumps(v, ensure_ascii=True)) for k, v in rows.items()))
            self.db.execute('INSERT OR REPLACE INTO metadata VALUES (?,?)', ('map_identity', encoded))
    def inventory_snapshot(self,garage):
        with self.db:
            self.db.execute('DELETE FROM garage_inventory'); self.db.execute('DELETE FROM garage_sections')
            self.db.executemany('INSERT INTO garage_inventory VALUES (?,?)',((k,json.dumps(v)) for k,v in garage.vehicles.items()))
            names=('garage','crew','depot','depot_details','research','progression','configured_performance','equipment_compatibility','equipment_removal_costs','crew_training_costs','discounts','account_extra','goodies','blueprints','current_shop_prices')
            self.db.executemany('INSERT INTO garage_sections VALUES (?,?)',((k,json.dumps(garage.sections[k])) for k in names if k in garage.sections))
    def reset(self):
        with self.db:
            self.db.execute('DELETE FROM garage_inventory'); self.db.execute('DELETE FROM garage_sections')
            self.db.execute('DELETE FROM snapshots'); self.db.execute('DELETE FROM map_records'); self.db.execute('DELETE FROM metadata')
    def close(self):
        self.db.close()


class UnavailableMapCatalogue:
    def __init__(self):
        self.rows={}; self.identity=None; self.version=None; self.status='Maps module unavailable'
    def load(self,conn,force=False): return False
    def resolve(self,context): return None
    def image_png(self,key): raise ModuleUnavailable('Maps module unavailable')


class Model:
    def __init__(self, root, runtime=None):
        self.root = root
        self.store = Store(root)
        self.runtime=runtime or ModuleManager(BASE,VERSION).start(activate=False)
        self.catalogue = self.runtime.factory('map_catalogue',root) or UnavailableMapCatalogue()
        self.garage_data=Snapshot(root,self.runtime)
        self.garage_signature=None
        self.sources = {k: self.store.get(k) for k in ('connector', 'diagnostics')}
        self.sources['battle_context'] = {}
        self.stamps = {}
        self.disk = {}
        self.errors = {}
        self.next_catalogue_check = 0
        self.state, self.status, self.context = 'missing', 'Connector missing', None
    def _read(self, key):
        path = self.root / 'connector' / (key + '.json')
        try:
            st = path.stat(); stamp = (st.st_size, st.st_mtime_ns)
            if self.stamps.get(key) == stamp and self.disk.get(key):
                return
            data = read_json(path)
            if key == 'diagnostics':
                if not isinstance(data.get('sections'), dict) or not isinstance(data.get('account'), dict):
                    raise ValueError('Invalid diagnostics structure')
            elif key == 'connector':
                if not isinstance(data.get('data'), dict) or not isinstance(data.get('session'), str):
                    raise ValueError('Invalid connector structure')
            elif not isinstance(data.get('session'), str) or not isinstance(data.get('state'), str):
                raise ValueError('Invalid battle context structure')
            self.sources[key] = data
            self.stamps[key] = stamp
            self.disk[key] = True
            self.errors.pop(key, None)
            if key != 'battle_context':
                try: self.store.put(key, data)
                except (sqlite3.Error, OSError) as exc: self.errors['database'] = str(exc)
        except (OSError, ValueError, TypeError, UnicodeError) as exc:
            self.disk[key] = False
            self.errors[key] = str(exc)
    def poll(self, force=False, now=None):
        for key in ('connector', 'battle_context', 'diagnostics'):
            self._read(key)
        conn, ctx = self.sources['connector'], self.sources['battle_context']
        self.state, self.status, self.context = connection(conn, ctx, self.disk.get('connector'), self.disk.get('battle_context'), now)
        tick = time.monotonic()
        changed = False
        if force or tick >= self.next_catalogue_check:
            self.next_catalogue_check = tick + 5
            try: changed = self.catalogue.load(conn, force=force)
            except Exception as exc:
                logging.exception('Map module data failure'); self.runtime.errors['maps']=str(exc); self.catalogue=UnavailableMapCatalogue(); changed=True
            if changed and self.catalogue.identity:
                try: self.store.maps(self.catalogue.rows, self.catalogue.identity)
                except (sqlite3.Error, OSError) as exc: self.errors['database'] = str(exc)
        signature=(self.stamps.get('diagnostics'),conn.get('session'),conn.get('game_version'),self.disk.get('diagnostics'))
        if signature!=self.garage_signature or force or changed:
            self.garage_signature=signature
            self.garage_data.refresh(self.sources['diagnostics'],conn,self.state)
            try: self.store.inventory_snapshot(self.garage_data)
            except (sqlite3.Error,OSError,ValueError) as exc: self.errors['inventory_database']=str(exc)
        self.garage_data.state=self.state
        return changed
    @property
    def account(self):
        return obj(self.sources['connector'].get('data')) or obj(self.sources['diagnostics'].get('account'))
    @property
    def selected(self):
        # A null selection during a transition must never default to another tank.
        c = self.sources['connector']; tank = obj(self.account.get('selected_tank'))
        if self.context and self.context.get('vehicle_id') is not None:
            v = next((r for r in self.garage if str(r.get('intCD')) == str(self.context['vehicle_id'])), {})
            if v:
                return {'tank_id': v.get('intCD'), 'name': v.get('userName'), 'tier': v.get('level'), 'combat_xp': v.get('combat_xp')}
            if str(tank.get('tank_id')) == str(self.context['vehicle_id']): return tank
            return {'tank_id': self.context['vehicle_id']}
        if tank: return tank
        if self.state in ('closed', 'stale', 'missing', 'version'):
            return obj(c.get('last_known_selected_tank'))
        return {}
    @property
    def garage(self):
        d = self.sources['diagnostics']; c = self.sources['connector']
        # On a fresh login do not blend older account snapshots with a new session.
        if c.get('session') and d.get('session') and c.get('session') != d.get('session'):
            return []
        items = obj(obj(d.get('sections')).get('garage')).get('data', [])
        return [obj(x.get('vehicle')) for x in items if isinstance(x, dict) and isinstance(x.get('vehicle'), dict)] if isinstance(items, list) else []
    def close(self): self.store.close()

REPO = 'itsr0b0tninja/WoTQuartermaster'
RELEASE_ROOT = 'https://github.com/' + REPO + '/releases/download/'
MANIFEST_URL = 'https://raw.githubusercontent.com/' + REPO + '/main/components.json'
COMPONENTS = {
    'desktop': ('WoTQuartermaster.exe', 'v', 180 * 1024 * 1024),
    'connector': ('wot_companion_connector.wotmod', 'connector-v', 10 * 1024 * 1024),
    'replay_parser': ('WoT_Quartermaster_Replay_Parser.txt', 'replay-v', 5 * 1024 * 1024),
    'tactics': ('tactics.json', 'tactics-v', 30 * 1024 * 1024),
}
HEX = re.compile(r'^[0-9a-f]{64}$')
VER = re.compile(r'^(0|[1-9]\d{0,3})\.(0|[1-9]\d{0,3})\.(0|[1-9]\d{0,3})$')
TOKEN = re.compile(r'^[0-9a-f]{32}$')
ALLOWED_HOSTS = {'github.com', 'raw.githubusercontent.com', 'release-assets.githubusercontent.com', 'objects.githubusercontent.com'}

def version_tuple(v):
    if not isinstance(v,str) or not VER.fullmatch(v): raise ValueError('Unsupported component version')
    return tuple(map(int,v.split('.')))

def check_url(url):
    p = urllib.parse.urlsplit(url)
    if p.scheme != 'https' or p.hostname not in ALLOWED_HOSTS or p.username or p.password or p.port not in (None,443):
        raise ValueError('Unexpected update download host')
    return url

class SafeRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return super().redirect_request(req,fp,code,msg,headers,check_url(newurl))

def open_url(url):
    req = urllib.request.Request(check_url(url),headers={'User-Agent':'WoTQuartermaster/'+VERSION,'Accept':'application/octet-stream'})
    return urllib.request.build_opener(SafeRedirect()).open(req,timeout=15)

def validate_manifest(raw):
    if not isinstance(raw,dict) or raw.get('schema') != 1 or not isinstance(raw.get('components'),dict):
        raise ValueError('Unsupported component manifest schema')
    result = {}
    for key, entry in raw['components'].items():
        if key not in COMPONENTS: continue
        if not isinstance(entry,dict): raise ValueError('Invalid component entry')
        name,prefix,limit = COMPONENTS[key]
        v = entry.get('version'); version_tuple(v)
        if entry.get('asset') != name or entry.get('release') != prefix+v or entry.get('url') != RELEASE_ROOT+prefix+v+'/'+name:
            raise ValueError('Unexpected '+key+' release location')
        check_url(entry['url'])
        if not isinstance(entry.get('sha256'),str) or not HEX.fullmatch(entry['sha256']):
            raise ValueError('Missing component checksum')
        if type(entry.get('size')) is not int or not 0 < entry['size'] <= limit:
            raise ValueError('Invalid component size')
        if entry.get('channel') not in ('alpha','stable'): raise ValueError('Unknown release channel')
        if entry.get('min_desktop') is not None: version_tuple(entry['min_desktop'])
        if key=='connector' and (entry.get('context_schema')!=1 or not isinstance(entry.get('game_versions'),list) or not entry['game_versions']):
            raise ValueError('Unsupported Connector compatibility metadata')
        result[key] = dict(entry)
    if 'desktop' not in result or 'connector' not in result: raise ValueError('Incomplete component manifest')
    return result

def fetch_manifest():
    with open_url(MANIFEST_URL) as response:
        raw = response.read(256*1024+1)
    if len(raw)>256*1024: raise ValueError('Manifest too large')
    raw=json.loads(raw.decode('utf-8'))
    entries,retired=validate_modules(raw)
    return validate_manifest(raw),entries,retired

def digest_file(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda:f.read(1024*1024),b''): h.update(block)
    return h.hexdigest()

def matches(path, entry):
    try: return path.is_file() and path.stat().st_size==entry['size'] and digest_file(path)==entry['sha256']
    except OSError: return False

_ui_hash_cache={}
def cached_match(path,entry):
    try:
        st=path.stat()
        key=(str(path),st.st_size,st.st_mtime_ns,entry['sha256'])
        if key not in _ui_hash_cache:
            if len(_ui_hash_cache)>100: _ui_hash_cache.clear()
            _ui_hash_cache[key]=matches(path,entry)
        return _ui_hash_cache[key]
    except (OSError,KeyError,TypeError): return False

def verify_payload(key,path):
    if key=='desktop':
        with path.open('rb') as f:
            if f.read(2)!=b'MZ': raise ValueError('Not a Windows executable')
    elif key=='connector':
        with zipfile.ZipFile(path) as z:
            expected='res/scripts/client/gui/mods/mod_wot_companion_connector.pyc'
            if z.namelist()!=[expected] or z.getinfo(expected).file_size>8*1024*1024 or z.testzip() is not None:
                raise ValueError('Invalid Connector archive')
            if z.read(expected)[:4]!=bytes.fromhex('03f30d0a'): raise ValueError('Unsupported Connector bytecode')
    elif key=='replay_parser':
        compile(path.read_text(encoding='utf-8'),str(path),'exec')
    elif key=='tactics':
        if not isinstance(read_json(path).get('schema'),int): raise ValueError('Invalid tactics schema')

def download_component(key,entry,dest,cancel,progress):
    # The manifest is validated even when the caller is the helper or a test.
    name,prefix,limit=COMPONENTS[key]
    if entry['url']!=RELEASE_ROOT+prefix+entry['version']+'/'+name: raise ValueError('Unexpected asset URL')
    dest.parent.mkdir(parents=True,exist_ok=True)
    tmp=dest.with_name(dest.name+'.part')
    if shutil.disk_usage(dest.parent).free < entry['size']*2+16*1024*1024: raise OSError('Insufficient free disk space')
    try:
        n=0; h=hashlib.sha256()
        with open_url(entry['url']) as response, tmp.open('wb') as f:
            while True:
                if cancel.is_set(): raise InterruptedError('Download cancelled')
                block=response.read(256*1024)
                if not block: break
                n+=len(block)
                if n>entry['size']: raise ValueError('Asset exceeds expected size')
                h.update(block); f.write(block); progress(n,entry['size'])
            f.flush(); os.fsync(f.fileno())
        if cancel.is_set(): raise InterruptedError('Download cancelled')
        if n!=entry['size'] or h.hexdigest()!=entry['sha256']: raise ValueError('Download size or SHA-256 mismatch')
        verify_payload(key,tmp); os.replace(tmp,dest)
    finally:
        tmp.unlink(missing_ok=True)

def game_version_evidence(game):
    """Return verified directory candidates and an installation fingerprint.
    Never infer the active version simply from the newest/only mods folder.
    """
    import xml.etree.ElementTree as ET
    game=game.resolve()
    mods=game/'mods'
    try:
        dirs=sorted(p.name for p in mods.iterdir() if p.is_dir() and re.fullmatch(r'\d+\.\d+\.\d+\.\d+',p.name) and p.resolve().parent==mods.resolve())
        file=game/'version.xml'
        raw=b''
        versions=set()
        if file.is_file():
            if file.stat().st_size>1024*1024: return None
            raw=file.read_bytes()
            root=ET.fromstring(raw)
            for item in root.iter():
                # Limit detection to the version text/attributes, not arbitrary numbers.
                tag=item.tag.rsplit('}',1)[-1].casefold()
                if 'version' in tag:
                    for value in [item.text or '',*item.attrib.values()]:
                        versions.update(re.findall(r'(?<!\d)(\d+\.\d+\.\d+\.\d+)(?![\d.])',value))
        fingerprint=hashlib.sha256(json.dumps([str(game),dirs,hashlib.sha256(raw).hexdigest()],ensure_ascii=True).encode('utf-8')).hexdigest()
        return dirs,versions,fingerprint
    except (OSError,ValueError,ET.ParseError): return None

def game_folder_version(game, reported='', remembered=None):
    evidence=game_version_evidence(game)
    if not evidence: return None
    dirs,versions,fingerprint=evidence
    exact=versions.intersection(dirs)
    m=re.search(r'(?<!\d)(\d+\.\d+\.\d+\.\d+)(?![\d.])',reported or '')
    live=m.group(1) if m else None
    if live:
        if live not in dirs: return None
        if exact and exact!={live}: return None
        # A version.xml containing the full build number rather than mod version
        # is acceptable only if a live Connector confirms the same patch family.
        if versions and not exact and not any(v.split('.')[:3]==live.split('.')[:3] for v in versions): return None
        return live
    if len(exact)==1: return next(iter(exact))
    if len(exact)>1: return None
    if remembered and remembered.get('root')==str(game.resolve()) and remembered.get('fingerprint')==fingerprint:
        v=remembered.get('version')
        if v in dirs and (not versions or any(x.split('.')[:3]==v.split('.')[:3] for x in versions)): return v
    return None

def component_status(local, entry, compatible=True):
    if not entry: return 'Not listed / not checked'
    if not compatible: return 'Incompatible with this desktop'
    if local is None: return 'Download available · not installed'
    if version_tuple(entry['version'])>version_tuple(local): return 'Update available'
    if version_tuple(entry['version'])==version_tuple(local): return 'Up to date'
    return 'Installed version is newer'

def connector_status(running, installed_verified, present, prepared, entry, compatible):
    if not entry:
        return 'Installed but unverified' if present else 'Not listed / not checked'
    current=running or installed_verified
    if not compatible:
        if current==entry['version']: return 'Up to date · destination compatibility not verified'
        return 'Game version or desktop compatibility not verified'
    if current:
        status=component_status(current,entry)
        if status=='Update available' and prepared: return 'Update downloaded · prepared for installation'
        return status
    if installed_verified: return component_status(installed_verified,entry)
    if present: return 'Installed but unverified'+(' · prepared for installation' if prepared else '')
    if prepared: return 'Prepared for installation'
    return 'Download available · not installed'

def write_connector_ini(folder,data):
    if not data.is_absolute() or not data.is_dir() or any(c in str(data) for c in '\r\n'):
        raise ValueError('Save a valid existing absolute data directory first')
    atomic_text(folder/'connector.ini','[Connector]\ndata_path='+str(data)+'\n')

    result=subprocess.run(['tasklist','/FO','CSV','/NH'],capture_output=True,text=True,errors='replace',timeout=8,creationflags=subprocess.CREATE_NO_WINDOW)
    if result.returncode: raise OSError('Could not verify whether World of Tanks is running')
    return any(row and row[0].casefold() in ('worldoftanks.exe','worldoftanks64.exe','worldoftanks32.exe') for row in csv.reader(result.stdout.splitlines()))

def update_paths(base):
    root=base/'updates'
    return root,root/'staging'/'WoTQuartermaster.exe',root/'backups'/'WoTQuartermaster.previous.exe'

def copy_atomic(source,dest):
    dest.parent.mkdir(parents=True,exist_ok=True)
    tmp=dest.with_name(dest.name+'.copying')
    try:
        shutil.copyfile(source,tmp)
        with tmp.open('rb+') as f: os.fsync(f.fileno())
        os.replace(tmp,dest)
    finally: tmp.unlink(missing_ok=True)

def helper_environment():
    env=os.environ.copy()
    env['PYINSTALLER_RESET_ENVIRONMENT']='1'
    return env

def wait_parent(pid):
    if os.name!='nt': raise OSError('Self-update requires Windows')
    import ctypes
    from ctypes import wintypes
    kernel=ctypes.WinDLL('kernel32',use_last_error=True)
    kernel.OpenProcess.argtypes=[wintypes.DWORD,wintypes.BOOL,wintypes.DWORD]; kernel.OpenProcess.restype=wintypes.HANDLE
    kernel.WaitForSingleObject.argtypes=[wintypes.HANDLE,wintypes.DWORD]; kernel.WaitForSingleObject.restype=wintypes.DWORD
    kernel.CloseHandle.argtypes=[wintypes.HANDLE]
    handle=kernel.OpenProcess(0x00100000,False,pid)
    if not handle:
        if ctypes.get_last_error()==87: return
        raise OSError('Cannot open prior desktop process')
    try:
        if kernel.WaitForSingleObject(handle,30000)!=0: raise OSError('Desktop did not close')
    finally: kernel.CloseHandle(handle)

def replace_retry(source,target):
    for attempt in range(30):
        try: copy_atomic(source,target); return
        except PermissionError:
            if attempt==29: raise
            time.sleep(.3)

def backup_info(base, cached=False):
    root,stage,backup=update_paths(base)
    entry=read_json(root/'backups'/'previous.json')
    version_tuple(entry['version'])
    if not (cached_match if cached else matches)(backup,entry): raise ValueError('Previous executable missing or damaged')
    return entry

def apply_update_helper(base,token,parent):
    root,stage,backup=update_paths(base)
    if not TOKEN.fullmatch(token): return 2
    target=base/'WoTQuartermaster.exe'
    request=root/('request-'+token+'.json'); health=root/('healthy-'+token+'.json')
    log=root/'update-result.json'
    proc=None; swapped=False; old_saved=False; parent_closed=False
    try:
        entry=read_json(request)
        if entry.get('token')!=token or entry.get('parent')!=parent: raise ValueError('Update request mismatch')
        version_tuple(entry['version']); version_tuple(entry['old_version'])
        wait_parent(parent); parent_closed=True
        if not matches(stage,entry): raise ValueError('Staged executable missing or damaged')
        verify_payload('desktop',stage)
        if not target.is_file() or game_running(): raise OSError('Close the game before applying updates')
        old={'version':entry['old_version'],'size':target.stat().st_size,'sha256':digest_file(target)}
        copy_atomic(target,backup); old_saved=True
        atomic_text(root/'backups'/'previous.json',json.dumps(old))
        replace_retry(stage,target); swapped=True
        proc=subprocess.Popen([str(target),'--update-health',token],cwd=str(base),env=helper_environment())
        deadline=time.monotonic()+100
        while time.monotonic()<deadline:
            if health.is_file():
                ack=read_json(health)
                if ack.get('token')==token and ack.get('version')==entry['version'] and proc.poll() is None:
                    atomic_text(log,json.dumps({'status':'success','version':entry['version']}))
                    stage.unlink(missing_ok=True); return 0
            if proc.poll() is not None: break
            time.sleep(.5)
        raise OSError('Updated desktop did not confirm successful startup')
    except Exception as exc:
        message=str(exc); recovery='not needed'
        if swapped and old_saved:
            try:
                if proc and proc.poll() is None:
                    subprocess.run(['taskkill','/PID',str(proc.pid),'/T','/F'],capture_output=True,timeout=15,creationflags=subprocess.CREATE_NO_WINDOW)
                    proc.wait(timeout=15)
                replace_retry(backup,target); recovery='previous executable restored'
            except Exception as rollback:
                recovery='automatic rollback failed: '+str(rollback)+'; backup retained at '+str(backup)
        try: atomic_text(log,json.dumps({'status':'failed','error':message,'recovery':recovery}))
        except OSError: pass
        try:
            if parent_closed and (not swapped or recovery=='previous executable restored'):
                # If parent never exited, do not start a duplicate instance.
                if target.is_file(): subprocess.Popen([str(target)],cwd=str(base),env=helper_environment())
        except OSError: pass
        return 1
    finally:
        request.unlink(missing_ok=True); health.unlink(missing_ok=True)

def start_helper(base,entry):
    if not getattr(sys,'frozen',False) or os.name!='nt': raise ValueError('Self-update requires the compiled Windows application')
    root,stage,backup=update_paths(base)
    target=base/'WoTQuartermaster.exe'
    if Path(sys.executable).resolve()!=target.resolve(): raise ValueError('Name the running executable WoTQuartermaster.exe before updating')
    if game_running(): raise ValueError('Close World of Tanks before installing the desktop update')
    if not matches(stage,entry): raise ValueError('Staged desktop checksum mismatch')
    token=secrets.token_hex(16)
    runner=root/('runner-'+token+'.exe')
    copy_atomic(target,runner)
    atomic_text(root/('request-'+token+'.json'),json.dumps(dict(entry,token=token,parent=os.getpid(),old_version=VERSION)))
    subprocess.Popen([str(runner),'--apply-update',token,str(os.getpid())],cwd=str(base),env=helper_environment(),creationflags=subprocess.CREATE_NO_WINDOW)

        def poll_updates(self):
            while True:
                try: kind,payload=self.update_queue.get_nowait()
                except queue.Empty: break
                if kind=='progress':
                    self.progress.setValue(int(payload[0]*100/payload[1])); continue
                self.update_busy=False; self.progress.hide(); self.cancel_button.hide(); self.check_updates_button.setEnabled(True)
                if kind=='error':
                    self.manual_module_request=None
                    self.update_status.setText('Update operation failed: '+payload+' · Existing components retained')
                elif payload[0]=='manifest':
                    self.update_entries,offered,retired=payload[1]
                    if self.settings.values['module_updates']!='manual' or self.manifest_manual:
                        self.module_entries,self.retired_modules=offered,retired
                    self.last_check_text=datetime.now().strftime('%d %b %H:%M'); self.manifest_cached=False
                    self.update_status.setText('Checked: '+self.last_check_text)
                    try: atomic_text(base/'updates'/'component-manifest.json',json.dumps({'schema':1,'components':self.update_entries,'modules':self.module_entries,'retired_modules':self.retired_modules}))
                    except OSError as exc: self.update_status.setText('Checked; offline cache not saved: '+str(exc))
                    request=self.manual_module_request; self.manual_module_request=None
                    if request or self.settings.values['module_updates']=='automatic': self.prepare_modules(*(request or (False,None)))
                elif payload[0]=='desktop':
                    self.pending_desktop=payload[1]; self.update_status.setText('Verified desktop downloaded. Close the game before installing.')
                    if (not self.model or self.model.state not in ACTIVE_BATTLE|{'matchmaking','waiting','stale'}) and QMessageBox.question(self,'Install desktop update','Restart WoTQuartermaster to install '+payload[1]['version']+'? Close World of Tanks first.')==QMessageBox.StandardButton.Yes: self.install_desktop(payload[1])
                else: self.update_status.setText(payload[1])
                self.refresh_update_rows()
            # Avoid hashing installed/prepared packages every 200 ms.
            if getattr(self,'update_refresh_at',0)<time.monotonic():
                self.update_refresh_at=time.monotonic()+5; self.refresh_update_rows()

                        copy_atomic(folder/'previous'/target.name,target); copy_atomic(folder/'previous'/'installed.json',prior)
                    raise

            self.metrics['combat_xp'].setText(number(selected.get('combat_xp'))); self.metrics['garage'].setText(number(len(m.garage)) if m.garage else '—')
            self.over_note.setText(('Saved account snapshot · ' if m.state in ('closed','stale','missing','version') else 'Account snapshot · ')+fmt_time(conn.get('sampled_at') or diag.get('sampled_at'))+('\nPremium account active' if acc.get('premium') is True else '\nStandard account' if acc.get('premium') is False else ''))
            self.runtime.dispatch('refresh',self,bool(force or cat_changed))
            self.module_status.setText(self.runtime.status_text())
            # Switch only when entering a different battle or leaving the battle group;
            # manual navigation is not overridden every poll. Never raise or activate.
            mode=('battle',m.context.get('battle_id') if m.context else None) if m.state in ACTIVE_BATTLE else ('garage',None) if m.state in ('garage','matchmaking') else ('offline',None)
            previous=self.last_auto
            if self.settings.flag('auto_switch') and mode!=previous:
                if mode[0]=='battle': self.navigate('Battle')
                elif previous and previous[0]=='battle': self.navigate('Overview')
            self.last_auto=mode
            self.footer.setText(f'Connector {conn.get("connector_version","—")}  ·  Game {conn.get("game_version","—")}  ·  {m.state.replace("_"," ")}  ·  Database schema {DB_SCHEMA}')
            game=absolute(base,self.settings.values['game_path']); errors='\n'.join(k+': '+str(v) for k,v in m.errors.items())
            self.diagnostic_label.setText('Data: '+str(m.root)+'\nGame folder: '+('found' if (game/'mods').is_dir() else 'not found or unrecognised')+'\n'+m.catalogue.status+'\n'+('Connector data may be incomplete.' if diag.get('status')=='partial' else 'Diagnostics: '+str(diag.get('status','—')))+('\n'+self.settings.warning if self.settings.warning else '')+('\n'+errors if errors else ''))
    # The activation journal remains provisional until the full initial UI
    # and model have been constructed and checked. On failure, rebuild with
    # the prior active set; an interrupted process is recovered at next launch.
    window=None
    try:
        window=Window()
        if manager.state['trial']:
            if not runtime.ready() or window.model is None: raise ValueError(runtime.status_text() or window.model_error)
            manager.confirm()
    except Exception as exc:
        if not manager.state['trial']: raise
        logging.exception('Module trial UI failed')
        manager.reject_trial(str(exc))
        app.setQuitOnLastWindowClosed(False)
        if window is not None: window.close(); window.deleteLater()
        runtime=manager.start(activate=False)
        window=Window(); app.setQuitOnLastWindowClosed(True)
    window.show()
    if test_callback is not None:
        def exercise():
            try: test_callback(window)
            except Exception: logging.exception('Module UI fixture failed'); app.exit(3); return
            window.close(); app.exit(0)
        QTimer.singleShot(0,exercise)
    if not health_token:
        for runner in (base/'updates').glob('runner-*.exe'):
            try: runner.unlink()
            except OSError: pass
    if health_token and TOKEN.fullmatch(health_token):
        def confirm_startup():
            if window.model is not None and window.isVisible() and runtime.ready():
                atomic_text(base/'updates'/('healthy-'+health_token+'.json'),json.dumps({'token':health_token,'version':VERSION}))
        QTimer.singleShot(3000,confirm_startup)
    try: return app.exec()
    finally: lock.unlock()


def main():
    parser=argparse.ArgumentParser(description='WoTQuartermaster portable alpha')
    parser.add_argument('--check-data',type=Path,help='Validate local data without opening Qt; no game modification')
    parser.add_argument('--apply-update',nargs=2,metavar=('TOKEN','PARENT'))
    parser.add_argument('--update-health')
    args=parser.parse_args()
    if args.apply_update:
        token,pid=args.apply_update
        if BASE.name!='updates' or not TOKEN.fullmatch(token) or not pid.isdigit(): return 2
        return apply_update_helper(BASE.parent,token,int(pid))
    if args.check_data is not None:
        model=Model(args.check_data.resolve())
        try:
            model.poll(True); print(json.dumps({'state':model.state,'status':model.status,'account':model.account,'garage_count':len(model.garage),'map_records':len(model.catalogue.rows),'catalogue_status':model.catalogue.status,'errors':model.errors},indent=2)); return 0
        finally: model.close()
    (BASE/'logs').mkdir(exist_ok=True)
    logging.basicConfig(filename=BASE/'logs'/'desktop.log',level=logging.INFO,format='%(asctime)s %(levelname)s %(message)s',encoding='utf-8')
    try: return run_gui(BASE,args.update_health)
    except Exception:
        logging.exception('Fatal desktop error'); raise

if __name__=='__main__':
    sys.exit(main())
