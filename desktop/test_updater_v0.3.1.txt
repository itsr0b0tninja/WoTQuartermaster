# Run using Python 3.12 from the repository root.
import importlib.machinery, importlib.util, io, hashlib, json, tempfile, threading, zipfile
from pathlib import Path
from unittest.mock import patch
loader=importlib.machinery.SourceFileLoader('quartermaster','desktop/WoTQuartermaster_v0.3.1.txt')
spec=importlib.util.spec_from_loader(loader.name,loader)
m=importlib.util.module_from_spec(spec); loader.exec_module(m)
def entry(key,v,data):
    name,prefix,_=m.COMPONENTS[key]
    return {'version':v,'release':prefix+v,'asset':name,'url':m.RELEASE_ROOT+prefix+v+'/'+name,'sha256':hashlib.sha256(data).hexdigest(),'size':len(data),'channel':'alpha','min_desktop':'0.2.0'}
def reject(fn):
    try: fn()
    except (ValueError,InterruptedError,OSError): return
    raise AssertionError('Invalid update accepted')
new=b'MZnew executable'; old=b'MZold executable'
d=entry('desktop','0.2.1',new)
c=entry('connector','0.0.8',b'zip'); c.update(context_schema=1,game_versions=['2.4.2.0'])
manifest={'schema':1,'components':{'desktop':d,'connector':c}}
assert m.validate_manifest(manifest)['desktop']==d
reject(lambda:m.validate_manifest({'schema':2,'components':manifest['components']}))
reject(lambda:m.validate_manifest({'schema':1,'components':dict(manifest['components'],desktop=dict(d,url='https://evil.example/a.exe'))}))
reject(lambda:m.validate_manifest({'schema':1,'components':dict(manifest['components'],desktop=dict(d,sha256='bad'))}))
assert m.version_tuple('0.10.0')>m.version_tuple('0.2.0')
reject(lambda:m.version_tuple('../1.0'))
with tempfile.TemporaryDirectory() as tmp:
    base=Path(tmp); dest=base/'download.exe'; dest.write_bytes(old)
    cancel=threading.Event()
    with patch.object(m,'open_url',lambda url:io.BytesIO(b'bad')):
        reject(lambda:m.download_component('desktop',d,dest,cancel,lambda a,b:None))
    assert dest.read_bytes()==old and not dest.with_name(dest.name+'.part').exists()
    cancel.set()
    with patch.object(m,'open_url',lambda url:io.BytesIO(new)):
        reject(lambda:m.download_component('desktop',d,dest,cancel,lambda a,b:None))
    assert dest.read_bytes()==old
    cancel.clear()
    with patch.object(m,'open_url',lambda url:io.BytesIO(new)):
        m.download_component('desktop',d,dest,cancel,lambda a,b:None)
    assert m.matches(dest,d)
    game=base/'game'; (game/'mods'/'2.4.2.0').mkdir(parents=True)
    assert m.game_folder_version(game) is None
    assert m.game_folder_version(game,'v.2.4.2.0 #1000')=='2.4.2.0'
    evidence=m.game_version_evidence(game)
    remembered={'root':str(game.resolve()),'version':'2.4.2.0','fingerprint':evidence[2]}
    assert m.game_folder_version(game,remembered=remembered)=='2.4.2.0'
    (game/'mods'/'2.3.0.0').mkdir()
    assert m.game_folder_version(game,remembered=remembered) is None
    assert m.game_folder_version(game) is None
    assert m.game_folder_version(game,'v.2.4.2.0 #1000')=='2.4.2.0'
    (game/'version.xml').write_text('<version>v.2.3.0.0 #1234</version>')
    assert m.game_folder_version(game)=='2.3.0.0'
    assert m.game_folder_version(game,'v.2.4.2.0 #1000') is None
    (game/'version.xml').write_text('<version>v.2.4.2.1578</version>')
    assert m.game_folder_version(game) is None
    assert m.game_folder_version(game,'v.2.4.2.0 #1000')=='2.4.2.0'
    (game/'version.xml').write_text('<version>v.2.4.2.0</version>')
    assert m.game_folder_version(game)=='2.4.2.0'
    assert m.game_folder_version(base/'not-a-game') is None
    assert m.component_status(None,d)=='Download available · not installed'
    assert m.component_status('0.2.0',d)=='Update available'
    assert m.component_status('0.2.1',d)=='Up to date'
    assert m.connector_status('0.0.8',None,True,True,c,True)=='Up to date'
    assert m.connector_status(None,None,True,True,c,True)=='Installed but unverified · prepared for installation'
    assert m.connector_status(None,None,False,True,c,True)=='Prepared for installation'
    assert m.connector_status(None,None,False,False,c,True)=='Download available · not installed'
    data=base/'data'; data.mkdir(); (data/'sentinel').write_text('preserve')
    m.write_connector_ini(base/'prepared',data)
    assert str(data) in (base/'prepared'/'connector.ini').read_text()
    settings=m.Settings(base); settings.values['playstyle']='Passive scout'; settings.save()
    saved=(base/'settings.ini').read_bytes()
    root,stage,backup=m.update_paths(base); stage.parent.mkdir(parents=True)
    target=base/'WoTQuartermaster.exe'; target.write_bytes(old)
    stage.write_bytes(new)
    token='a'*32; parent=123
    request=root/('request-'+token+'.json')
    request.write_text(json.dumps(dict(d,token=token,parent=parent,old_version='0.2.0')))
    calls=[]
    class Failed:
        pid=456
        def poll(self): return 1
    def popen(*a,**kw): calls.append(a); return Failed()
    with patch.object(m,'wait_parent',lambda p:None),patch.object(m,'game_running',lambda:False),patch.object(m.subprocess,'Popen',popen):
        assert m.apply_update_helper(base,token,parent)==1
    assert target.read_bytes()==old and backup.read_bytes()==old
    assert m.read_json(root/'update-result.json')['recovery']=='previous executable restored'
    assert (base/'settings.ini').read_bytes()==saved and (data/'sentinel').read_text()=='preserve'
    token='b'*32; stage.write_bytes(new); (root/('request-'+token+'.json')).write_text(json.dumps(dict(d,token=token,parent=parent,old_version='0.2.0')))
    class Alive:
        pid=456
        def poll(self): return None
    def healthy(*a,**kw):
        (root/('healthy-'+token+'.json')).write_text(json.dumps({'token':token,'version':'0.2.1'}))
        return Alive()
    with patch.object(m,'wait_parent',lambda p:None),patch.object(m,'game_running',lambda:False),patch.object(m.subprocess,'Popen',healthy):
        assert m.apply_update_helper(base,token,parent)==0
    assert target.read_bytes()==new and backup.read_bytes()==old and not stage.exists()
    assert (base/'settings.ini').read_bytes()==saved and (data/'sentinel').read_text()=='preserve'
print('Manifest, semantic versions, unsafe URLs, corrupt download, cancellation, verified download, game path, INI, simulated replacement and rollback, data preservation: passed')
